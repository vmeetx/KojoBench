<geometry>
Goal: Draw three regular hexagons arranged in a triangular pattern.
Canvas/scale: 500x500 pixels.
Main shapes:
- Three hexagons, each with side length 100 pixels.
Drawing order:
1. First hexagon at the bottom center.
2. Second hexagon above and to the left of the first.
3. Third hexagon above and to the right of the first.
</geometry>

```scala
clear()
setSpeed(fast)

// Draw the first hexagon at the bottom center
def drawHexagon(sideLength: Double) {
  repeat(6) {
    forward(sideLength)
    right(60)
  }
}

drawHexagon(100)

// Move to the position for the second hexagon
penUp()
forward(50 * math.sqrt(3))
right(60)
forward(50)
left(60)
penDown()

// Draw the second hexagon
drawHexagon(100)

// Move to the position for the third hexagon
penUp()
back(100)
right(60)
forward(50 * math.sqrt(3))
left(60)
penDown()

// Draw the third hexagon
drawHexagon(100)
```
